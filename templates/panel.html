function renderAccess(host,key,type,revealed){
  const shownKey=revealed?key:'••••••••••••••••••••••••';
  const sub = host.split('.')[0];
  const nic = 'https://'+ENDPOINT+'/nic/update';

  const family=type==='AAAA'?'-6':'-4';
  const curlOne = 'curl '+family+' -u '+host+':'+shownKey+' '+nic;
  const dockerDdns =
    'docker run -d --name ddns-'+sub+' --restart=always \\\n'+
    '  curlimages/curl sh -c \\\n'+
    "  'while true; do curl "+family+" -su "+host+':'+shownKey+' '+nic+'; sleep 300; done\'';
  document.getElementById('pane-ddns').innerHTML =
    codeBlock('① '+tt('js_c_test'), curlOne, curlOne) +
    codeBlock('② '+tt('js_c_docker_ddns'), dockerDdns, dockerDdns);

  const ddc = 'protocol=dyndns2\nuse=web\nserver='+ENDPOINT+'\nssl=yes\nlogin='+host+'\npassword='+shownKey+'\n'+host;
  document.getElementById('pane-ddclient').innerHTML = codeBlock('/etc/ddclient.conf', ddc, ddc);

  document.getElementById('snip-router').innerHTML =
    '# '+tt('js_router_comment')+'\n'+
    tt('js_server')+'   : '+ENDPOINT+'\n'+
    tt('js_hostname')+' : '+host+'\n'+
    tt('js_username')+' : '+host+'\n'+
    tt('js_password')+' : '+shownKey+'';
}

async function verifyToken(id){
  const el=document.getElementById('vs-'+id);
  setStat(el,'run',SPIN);
  try{
    const r=await fetch('/tokens/'+id+'/verify');const d=await r.json();
    if(d.ok){ setStat(el,'ok',OKI+tt('js_valid')+' ('+d.status+')'); pop(tt('js_token_ok',{n:(d.zones||[]).length})); }
    else{ setStat(el,'bad',BADI+(d.error||tt('js_invalid'))); }
  }catch(e){ setStat(el,'bad',BADI+tt('js_req_fail')); }
  setTimeout(()=>{el.style.opacity='0';setTimeout(()=>{el.innerHTML='';el.style.opacity='1';el.className='vstat';},400);},6000);
}

function termLine(text){
  const div=document.createElement('span'); div.className='l';
  let cls='';
  if(text.startsWith('$'))cls='prompt'; else if(text.startsWith('▶'))cls='step';
  else if(text.startsWith('✔'))cls='okl'; else if(text.startsWith('✖')||text.includes('[timeout'))cls='errl';
  if(cls)div.classList.add(cls);
  div.textContent=text||' ';
  return div;
}
async function issueCert(certId,target, idx,reloadPage=true){
  const body=document.getElementById('term-body');
  const badge=document.getElementById('term-badge');
  const closeBtn=document.getElementById('term-close');
  document.getElementById('term-title').textContent='acme.sh — '+target;
  body.innerHTML=''; badge.className='term-badge run'; badge.innerHTML=''+tt('js_running');
  closeBtn.disabled=true; closeBtn.textContent=tt('js_running_dots');
  const caret=document.createElement('span'); caret.className='term-caret'; body.appendChild(caret);
  document.getElementById('termModal').showModal();
  let jid;
  try{
    const r=await fetch('/certs/'+encodeURIComponent(certId)+'/issue',{method:'POST'});
    const d=await r.json();if(!r.ok||!d.job)throw new Error(d.error||tt('js_start_fail'));jid=d.job;
  }catch(e){ badge.className='term-badge err'; badge.textContent=tt('js_start_fail'); closeBtn.disabled=false; closeBtn.textContent=tt('js_close'); return; }

  let cursor=0;
  return new Promise(resolve=>{
  const poll=async()=>{
    try{
      const r=await fetch('/certs/job/'+jid+'?from='+cursor);
      const d=await r.json();
      if(!r.ok)throw new Error(d.error||tt('js_req_fail'));
      if(d.lines && d.lines.length){
        d.lines.forEach(t=>body.insertBefore(termLine(t), caret));
        cursor=d.next; body.scrollTop=body.scrollHeight;
      }
      if(d.done){
        caret.remove();
        if(d.ok){ badge.className='term-badge ok'; badge.innerHTML=OKI+tt('js_success'); pop(tt('js_cert_done',{zone:target})); }
        else{ badge.className='term-badge err'; badge.innerHTML=BADI+tt('js_fail'); }
        closeBtn.disabled=false; closeBtn.textContent=tt('js_close');
        if(d.ok&&reloadPage) setTimeout(()=>location.reload(),1800);
        resolve(d.ok);
        return;
      }
    }catch(e){caret.remove();badge.className='term-badge err';badge.textContent=tt('js_req_fail');closeBtn.disabled=false;closeBtn.textContent=tt('js_close');resolve(false);return;}
    setTimeout(poll, 700);
  };
  poll();
  });
}
